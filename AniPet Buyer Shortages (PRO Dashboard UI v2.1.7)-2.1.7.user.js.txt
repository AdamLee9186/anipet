          badge(s.recommend_sku, "gray"), badge(t(s.category_key), "gray")
        ]),
        el("div", { style: { display: "flex", alignItems: "center", gap: "8px", flexWrap: "wrap" } }, badgeList)
      ]);

      const btnToggle = el("button", { className: "ani-btn", textContent: "פתח לקוחות" });
      const cBody = el("div", { className: "ani-card-body" });
      let isLoaded = false;

      btnToggle.onclick = async () => {
        const isOpen = cBody.classList.contains('open');
        if (isOpen) {
            cBody.classList.remove('open');
            btnToggle.textContent = "פתח לקוחות";
        } else {
            cBody.classList.add('open');
            btnToggle.textContent = "סגור רשימה";
            if (!isLoaded) {
                cBody.innerHTML = "";
                cBody.appendChild(createSkeletonTable());
                await fetchAndRenderCustomers(s.recommend_sku, cBody);
                isLoaded = true;
            }
        }
      };

      card.appendChild(el("div", { className: "ani-card-header" }, [titleArea, btnToggle]));
      card.appendChild(cBody);
      mainList.appendChild(card);
    }
  }

  async function fetchAndRenderCustomers(sku, containerElement) {
    try {
      const users = await sbRpc("buyer_sku_shortages_drilldown_step1_base", { p_recommend_sku: sku, p_category_key: null, p_limit: state.limit, p_grace_past_days: state.graceDays, p_window_days: state.windowDays });
      containerElement.innerHTML = "";

      if (!users.length) { containerElement.appendChild(el("div", { textContent: "לא נמצאו לקוחות מתאימים.", style: { padding: "16px", textAlign: "center", color: "#94a3b8" } })); return; }

      const phones = users.map(u => u.p9);
      const cat = users[0].category_key;
      let nameMap = {}, ordersMap = {};
      try {
          const namesRaw = await sbRpc("buyer_sku_shortages_raw_names", { p_phones: phones });
          for (const n of namesRaw) nameMap[n.p9] = { name: n.c_name, addr: [n.c_city, n.c_address].filter(Boolean).join(" - ") };
      } catch(e) {}

      try {
          const orderPromises = phones.map(phone => sbRpc("buyer_sku_shortages_raw_orders_single", { p_phone: phone, p_category: cat }).then(res => res.map(r => ({ p9: phone, o_date: r.o_date, qty: r.qty }))).catch(()=>[]));
          const ordersRaw = (await Promise.all(orderPromises)).flat();
          for (const o of ordersRaw) { if (!ordersMap[o.p9]) ordersMap[o.p9] = []; ordersMap[o.p9].push({ date: new Date(o.o_date), qty: Number(o.qty) }); }
      } catch(e) {}

      const custRows = users.map(u => {
        const p9 = u.p9;
        let uOrders = ordersMap[p9] || []; uOrders.sort((a,b)=>b.date-a.date);

        let anomalyBadge = badge("יציב", "gray"), cv = null, relBadge = badge("מעט נתונים", "gray");
        let lastDate = null, lastQty = 0;

        if (uOrders.length > 0) {
            lastDate = uOrders[0].date.toISOString().split('T')[0];
            lastQty = uOrders[0].qty;
            let past = uOrders.slice(1,4);
            if (past.length > 0) {
                let avg = past.reduce((s,o)=>s+o.qty,0) / past.length;
                let diff = lastQty - avg;
                if (lastQty <= avg * 0.85) anomalyBadge = badge(`חסר ${Math.abs(diff).toFixed(1)}`, "red", `קנה ${lastQty} במקום ממוצע ${avg.toFixed(1)}`);
                if (lastQty >= avg * 1.15) anomalyBadge = badge(`עודף ${Math.abs(diff).toFixed(1)}`, "green", `קנה ${lastQty} במקום ממוצע ${avg.toFixed(1)}`);
            }

            let rates = [];
            for(let i=0; i<uOrders.length-1; i++) {
                let d = (uOrders[i].date - uOrders[i+1].date)/86400000;
                if (d>4) rates.push(uOrders[i+1].qty/d);
            }
            if (rates.length >= 2) {
                let mean = rates.reduce((a,b)=>a+b,0)/rates.length;
                let v = rates.reduce((a,b)=>a+Math.pow(b-mean,2),0)/rates.length;
                cv = mean > 0 ? Math.sqrt(v)/mean : 0;
                let cvPct = (cv*100).toFixed(1)+"%";
                if (cv <= 0.15) relBadge = badge("🎯 גבוהה", "green", `סטיית צריכה: ${cvPct}`);
                else if (cv <= 0.35) relBadge = badge("🟡 בינונית", "orange", `סטיית צריכה: ${cvPct}`);
                else relBadge = badge("🎲 נמוכה", "red", `סטיית צריכה: ${cvPct}`);
            }
        }

        let statusType = u.dos_bucket === "CRIT" ? "red" : (u.dos_bucket === "LOW" ? "orange" : (u.dos_bucket === "OK" ? "green" : "blue"));
        const infoBtn = el("button", { className: "ani-btn", textContent: "👁️", style: { padding: "4px 8px", fontSize: "14px", border: "none", background: "#f1f5f9" } });
        const cName = nameMap[p9]?.name || "לקוח לא ידוע";

        infoBtn.onclick = async () => {
            infoBtn.textContent = "⏳";
            try {
                const data = await sbRpc("get_customer_recent_items_fast", {
                    p_phone: p9,
                    p_category: cat
                });

                if (!data || !data.length) {
                    alert("לא נמצאו פריטים קודמים בקטגוריה זו.");
                    return;
                }

                const histRows = data.map(d => [fmtDate(d.order_date), d.item_name, d.qty || 1, `${d.qty_pack} ק"ג`]);
                const histTable = table(["תאריך", "פריט", "יחידות", 'סה"כ משקל'], histRows);

                openModal(`היסטוריית הזמנות של ${cName}`, el("div", {className: "table-responsive", style:{padding:0}}, [histTable]));
            } catch(err) {
                const errBody = err.body ? (typeof err.body === 'string' ? err.body : JSON.stringify(err.body)) : '';
                alert(`שגיאת שליפה מפורטת:\nסטטוס: ${err.status || 'לא ידוע'}\n${errBody || err.message}`);
            } finally {
                infoBtn.textContent = "👁️";
            }
        };

        const dateCell = el("div", { style: { display: "flex", alignItems: "center", gap: "8px" } }, [ el("span", { textContent: fmtDate(lastDate) }), infoBtn ]);
        const historyText = uOrders.length > 0 ? `${u.order_days_180d || 0} (סה"כ: ${uOrders.length})` : String(u.order_days_180d || 0);

        return [
          u.customer_phone || "", cName, nameMap[p9]?.addr || "",
          asNode(badge(t(u.category_key), "gray")), asNode(dateCell), asNode(anomalyBadge), asNode(relBadge),
          asNode(badge(t(u.dos_bucket), statusType)), fmtDate(u.next_expected_date), historyText
        ];
      });

      const relHeader = el("span", { className: "has-tooltip", 'data-tooltip': "מחושב עפ״י יציבות צריכה יומית לאורך זמן", textContent: "אמינות ❓" });
      const tblContainer = el("div", { className: "table-responsive" });
      tblContainer.appendChild(table(["טלפון", "שם לקוח", "כתובת למשלוח", "קטגוריה", "הזמנה אחרונה", "מגמת כמות", asNode(relHeader), "מלאי בבית", "צפי סיום", "הזמנות (חצי שנה)"], custRows));
      containerElement.appendChild(tblContainer);

    } catch(e) { containerElement.innerHTML = ""; containerElement.appendChild(el("div", { textContent: `שגיאה בשליפת לקוחות: ${e.message}`, style: { padding: "16px", color: "red" } })); }
  }

  loadDataAndRender();

})();
